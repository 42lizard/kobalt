cmake_minimum_required(VERSION 3.22)
project(KobaltPocketBook LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
option(KOBALT_POCKETBOOK_HOST "Build the PocketBook host for desktop checks" OFF)
option(KOBALT_POCKETBOOK_ARM_TEST "Build the reading test for firmware runtime checks under QEMU" OFF)
get_filename_component(KOBALT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(SRC "${KOBALT_ROOT}/src")
set(KOBALT_QT_MAJOR 6)
set(KOBALT_NO_SENSORS ON)
if(NOT DEFINED KOBALT_VERSION)
    file(STRINGS "${KOBALT_ROOT}/VERSION" KOBALT_VERSION LIMIT_COUNT 1)
endif()
if(KOBALT_POCKETBOOK_HOST)
    find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network Test)
else()
    if(NOT POCKETBOOK_QT_SDK)
        set(POCKETBOOK_QT_SDK "$ENV{POCKETBOOK_QT_SDK}")
    endif()
    set(POCKETBOOK_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
    add_subdirectory("${POCKETBOOK_QT_SDK}" sdk EXCLUDE_FROM_ALL)
    # Use the target firmware's Qt at link time. Debian Qt imports HarfBuzz
    # symbols that conflict with CREngine's private static HarfBuzz.
    foreach(module Core Gui Widgets Network)
        set(library "${KOBALT_FIRMWARE_LIBS}/libQt6${module}.so.6.8.2")
        if(NOT EXISTS "${library}")
            message(FATAL_ERROR "Set KOBALT_FIRMWARE_LIBS to the extracted PB743K3 Qt 6.8.2 library directory")
        endif()
        set_property(TARGET Qt6::${module} PROPERTY IMPORTED_LOCATION "${library}")
    endforeach()
    set_property(TARGET Qt6::Gui APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Core)
    set_property(TARGET Qt6::Widgets APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Gui Qt6::Core)
    set_property(TARGET Qt6::Network APPEND PROPERTY INTERFACE_LINK_LIBRARIES Qt6::Core)
    # The PocketBook SDK supplies imported targets without OpenGL or zstd.
    function(qt_add_resources target name)
        cmake_parse_arguments(ARG "" "PREFIX;BASE" "FILES" ${ARGN})
        set(xml "<RCC><qresource prefix=\"${ARG_PREFIX}\">")
        foreach(path IN LISTS ARG_FILES)
            file(RELATIVE_PATH alias "${ARG_BASE}" "${path}")
            file(RELATIVE_PATH source "${CMAKE_CURRENT_BINARY_DIR}" "${path}")
            string(APPEND xml "<file alias=\"${alias}\">${source}</file>")
        endforeach()
        string(APPEND xml "</qresource></RCC>")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.qrc" "${xml}")
        pocketbook_compile_qrc(resource "${CMAKE_CURRENT_BINARY_DIR}/${name}.qrc")
        target_sources(${target} PRIVATE "${resource}")
    endfunction()
endif()
include("${KOBALT_ROOT}/cmake/Strings.cmake")
add_compile_options("-iquote${SRC}" -Wall -Wextra)
set(KOBALT_HOST_SOURCES "${SRC}/platform/desktop/services.cpp"
    "${SRC}/platform/pocketbook/window.cpp" "${SRC}/platform/pocketbook/device.cpp")
include("${KOBALT_ROOT}/cmake/StandaloneReader.cmake")
if(KOBALT_POCKETBOOK_HOST)
    target_compile_definitions(standalone-runtime PUBLIC KOBALT_POCKETBOOK_HOST)
else()
    target_link_libraries(standalone-runtime PUBLIC PocketBook::InkView Qt6::Core)
endif()
add_executable(kobalt "${SRC}/platform/pocketbook/main.cpp")
target_link_libraries(kobalt PRIVATE standalone-runtime)
if(NOT KOBALT_POCKETBOOK_HOST)
    # Qt uses firmware FreeType, while CREngine retains its hidden static copy.
    # Make the DSO explicit so the linker does not bind Qt to hidden symbols.
    target_link_libraries(kobalt PRIVATE "${POCKETBOOK_TARGET_LIB_DIR}/libfreetype.so.6")
endif()
file(GLOB notices "${KOBALT_ROOT}/resources/notices/*")
qt_add_resources(kobalt notices PREFIX "/kobalt" BASE "${KOBALT_ROOT}/resources"
    FILES "${KOBALT_ROOT}/resources/acknowledgments.json" ${notices})
if(KOBALT_POCKETBOOK_ARM_TEST AND NOT KOBALT_POCKETBOOK_HOST)
    add_library(Qt6::Test SHARED IMPORTED)
    set_target_properties(Qt6::Test PROPERTIES
        IMPORTED_LOCATION "/usr/lib/arm-linux-gnueabi/libQt6Test.so.6"
        INTERFACE_COMPILE_DEFINITIONS "QT_TESTLIB_LIB;QT_WIDGETS_LIB"
        INTERFACE_INCLUDE_DIRECTORIES "/usr/include/arm-linux-gnueabi/qt6/QtTest")
endif()
if(KOBALT_POCKETBOOK_HOST OR KOBALT_POCKETBOOK_ARM_TEST)
    enable_testing()
    add_executable(pocketbook-test "${KOBALT_ROOT}/tests/pocketbook/window_test.cpp")
    target_link_libraries(pocketbook-test PRIVATE standalone-runtime Qt6::Test)
    if(NOT KOBALT_POCKETBOOK_HOST)
        target_link_libraries(pocketbook-test PRIVATE "${POCKETBOOK_TARGET_LIB_DIR}/libfreetype.so.6")
    endif()
    target_compile_definitions(pocketbook-test PRIVATE KOBALT_ASSETS="${KOBALT_ASSETS}" KOBALT_FIXTURE="${KOBALT_FIXTURE}")
    add_test(NAME pocketbook-reading COMMAND pocketbook-test)
    set_tests_properties(pocketbook-reading PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_LOGGING_RULES=*.debug=false")
endif()
